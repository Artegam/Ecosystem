306 void ScreenViews::NCurses::gameplay () {
307 
308  unsigned int const OCEAN = 0;
309  unsigned int const PLAIN = 1;
310  unsigned int const xOffset = 1;
311  unsigned int const yOffset = 1;
312  map<int, int> worldMap = data->getWorldData().getWorldMap();
313  unsigned int size = worldMap.size();
314  map<string, string> dict = data->getDictionary();
315 
316  if (toClear) {
317  clear();
318  toClear = false;
319  }
320 
321  windows[IN_GAME] = subwin(stdscr, worldHeight + 2, worldWidth + 2, 1, 0);
322  // dessin du bord de la fenetre
323  box(windows[IN_GAME], ACS_VLINE, ACS_HLINE);
324 
325  // Drawing the map
326  for(unsigned int index = 0; index < size; index++) {
327  int x = xOffset + (index % worldWidth);
328  int y = yOffset + ((index - (x - xOffset)) / worldWidth);
329 
330  if(worldMap[index] == OCEAN) {
331  drawChar (windows[IN_GAME], x, y, dict["Water"].c_str()[0], WATER_PAIR);
332  } else if (worldMap[index] == PLAIN) {
333  drawChar (windows[IN_GAME], x, y, dict["Plain"].c_str()[0], PLAIN_PAIR);
334  } else {
335  drawChar (windows[IN_GAME], x, y, dict["Empty"].c_str()[0], EMPTY_PAIR);
336  }
337  }
338 
339  //On affiche la faune
340  list<Wildlife *>::iterator it;
341  list<Wildlife *> wildlifes = data->getWildlife();
342  for(it = wildlifes.begin(); it != wildlifes.end(); it++) {
343  char type = (*it)->getDisplayChar();
344  int x = (*it)->getX()+1;
345  int y = (*it)->getY()+1;
346 
347  if(type == 'F') {
348  drawChar (windows[IN_GAME], x, y, type, FISH_PAIR);
349  } else if(type == 'S') {
350  drawChar (windows[IN_GAME], x, y, type, SHARK_PAIR);
351  } else {
352  drawChar (windows[IN_GAME], x, y, type, EMPTY_PAIR);
353  }
354  }
355 
356  refresh();
357  wrefresh(windows[IN_GAME]);
358  wmove(windows[ROOT], 0, 50); // repositione le curseur
359  wrefresh(windows[ROOT]);
360  usleep(200000);
361  // fin dessin de la fenetre
362 }
